\newcommand{\VSrad}[3]{%
  \sqrt{\frac{2#2\log(6/#3)}{#1}}
  +\frac{7\log(6/#3)}{3(#1-1)}%
}

For family $i$, level $h$, cell $I$, and checkpoint $r$, define
\[
\eta_{i,h,I,r}
:=
\frac{36\delta_i}
{\pi^4\,2^{d_i h}(h+1)^2(r+1)^2}.
\]
For $n\ge2$, write
\[
\operatorname{rad}(n,V,\eta)
:=
\sqrt{\frac{2V\log(6/\eta)}{n}}
+
\frac{7\log(6/\eta)}{3(n-1)}.
\]

\begin{algorithm}[t]
\caption{Resolve one dyadic level}
\label{alg:vs-resolve-level}
\small
\begin{algorithmic}[1]
\Require depth $h$ and the current active sets $\{\mathcal C_i(h)\}_{i=1}^K$
\Ensure \textsc{Success}, or \textsc{Fail} if the calendar cutoff is reached

\State $\rho_h\gets2^{-h-1}$
\For{$i=1,\ldots,K$}
    \State $a_{i,h}\gets\min\{1,\mathcal L_i^g\rho_h\}$
    \State mark every $I\in\mathcal C_i(h)$ unresolved
\EndFor
\State $r\gets0$

\While{some active cell is unresolved}
    \State $n_r\gets2^{r+1}$ \Comment{next geometric checkpoint}

    \For{each unresolved pair $(i,I)$ in fixed round-robin order}
        \While{$m_I<n_r$}
            \If{$t=B$}
                \State \Return \textsc{Fail}
            \EndIf
            \State deploy $(i,c_I)$ as a designated source at round $t+1$
            \State $\mathcal S_{\rm cert}\gets
            \mathcal S_{\rm cert}\cup\{t+1\}$
            \State $m_I\gets m_I+1$, $t\gets t+1$
        \EndWhile
    \EndFor

    \State freeze the active sets for the maturation flush
    \For{$u=1,\ldots,w$}
        \If{$t=B$}
            \State \Return \textsc{Fail}
        \EndIf
        \State deploy the first active center under the fixed deterministic order
        \State tag the filler source as certification-owned and exclude its feedback
        from all designated estimators
        \State $\mathcal S_{\rm cert}\gets
        \mathcal S_{\rm cert}\cup\{t+1\}$, $t\gets t+1$
    \EndFor

    \State finalize every newly generated designated source as
    $B_s^{(w)}\gets\mathbf 1\{Z_s=1,\ D_s\le w\}$

    \For{each still-unresolved pair $(i,I)$}
        \State $n_I\gets m_I$ and recompute $\widehat g_I$ and $V_I$
        from finalized designated observations at $c_I$
        \State $\operatorname{rad}_I\gets
        \VSrad{n_I}{V_I}{\eta_{i,h,I,r}}$
        \If{$\operatorname{rad}_I\le a_{i,h}/8$}
            \State mark $I$ resolved and freeze its statistics for level $h$
        \EndIf
    \EndFor

    \State $r\gets r+1$
\EndWhile
\State \Return \textsc{Success}
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[t]
\caption{Construct a within-family certificate}
\label{alg:vs-family-certificate}
\small
\begin{algorithmic}[1]
\Require family $i$, level $h$, and resolved active set $\mathcal C_i(h)$
\Ensure recommendation $z_i$, bounds $\underline M_i^g,\overline M_i^g$,
and error certificate $\xi_i^g$

\For{each $I\in\mathcal C_i(h)$}
    \State $\mathrm{LCB}(I)\gets
    \max\{0,\widehat g_I-\operatorname{rad}_I\}$
    \State $U_{\rm cell}(I)\gets
    \min\{1,\widehat g_I+\operatorname{rad}_I+a_{i,h}\}$
\EndFor

\State $\underline M_i^g\gets
\max_{I\in\mathcal C_i(h)}\mathrm{LCB}(I)$
\State $\overline M_i^g\gets
\max_{I\in\mathcal C_i(h)}U_{\rm cell}(I)$
\State choose $I_i^L\in\arg\max_{I\in\mathcal C_i(h)}\mathrm{LCB}(I)$
using fixed tie-breaking
\State $z_i\gets c_{I_i^L}$
\State $\xi_i^g\gets
\min\{1,\max\{0,\overline M_i^g-\underline M_i^g\}\}$
\State \Return $(z_i,\underline M_i^g,\overline M_i^g,\xi_i^g)$
\end{algorithmic}
\end{algorithm}
