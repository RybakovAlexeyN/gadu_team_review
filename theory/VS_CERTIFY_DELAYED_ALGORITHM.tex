% VS-Certify-Delayed: reader-facing paper pseudocode.
%
% Required packages:
%   \usepackage{amsmath,amssymb}
%   \usepackage{algorithm}
%   \usepackage{algpseudocode}
%
% The main procedure is intentionally short. The two helper procedures
% expose the statistical and geometric details without hiding them.

\algrenewcommand\algorithmicrequire{\textbf{Input:}}
\algrenewcommand\algorithmicensure{\textbf{Output:}}

\newcommand{\VSrad}[3]{%
  \sqrt{\frac{2#2\log(6/#3)}{#1}}
  +\frac{7\log(6/#3)}{3(#1-1)}%
}

\begin{algorithm}[t]
\caption{\textsc{VS-Certify-Delayed}}
\label{alg:vs-certify-delayed}
\small
\begin{algorithmic}[1]

\Require
families \(i=1,\ldots,K\) on \(X_i=[0,1]^{d_i}\);
valid Lipschitz bounds \(L_i>0\);
common known \(q_w>0\);
window \(w\in\mathbb Z_{\ge0}\);
risks \((\delta_i)_{i=1}^K\) with
\(\sum_i\delta_i\le\delta_{\rm cert}\);
calendar cutoff \(B\).

\Ensure
\(\textsc{Certified}
\bigl(i,\{z_j,\ell_j,U_j,\Xi_j\}_{j=1}^K,\mathsf{Acct}\bigr)\)
or
\(\textsc{Not-Certified}(\mathsf{Acct})\).

\Statex
\(\mathcal L_i^g\gets q_wL_i\),
\(\rho_h\gets2^{-h-1}\),
\[
\eta_{i,h,I,r}
:=
\frac{36\delta_i}
{\pi^4\,2^{d_i h}(h+1)^2(r+1)^2}.
\]
\Statex
\textbf{Global state:}
calendar time \(t\), owned source set \(\mathcal S_{\rm cert}\),
and per-cell statistics
\((m_I,n_I,\widehat g_I,V_I,\operatorname{rad}_I)\).

\State \(t\gets0\), \(h\gets0\),
       \(\mathcal S_{\rm cert}\gets\varnothing\).
\For{each family \(i\)}
    \State \(\mathcal C_i(0)\gets\{X_i\}\) with fresh root-cell state.
\EndFor

\While{true}
    \If{\(\Call{ResolveLevel}{h}=\textsc{Fail}\)}
        \State \Return \(\textsc{Not-Certified}(\Call{Acct}{t})\)
    \EndIf

    \For{each family \(i\)}
        \State
        \((z_i,\underline M_i^g,\overline M_i^g,\xi_i^g)
        \gets \Call{FamilyCertificate}{i,h}\)
    \EndFor

    \If{some \(i\) satisfies
        \(\underline M_i^g>
        \max_{j\ne i}\overline M_j^g\)}
        \State choose the first such \(i\) under the fixed family order
        \For{each family \(j\)}
            \State
            \(\ell_j\gets\max\{0,\underline M_j^g/q_w\}\)
            \State
            \(U_j\gets\min\{1,\overline M_j^g/q_w\}\)
            \State
            \(\Xi_j\gets\min\{1,\max\{0,\xi_j^g/q_w\}\}\)
        \EndFor
        \State \Return
        \(\textsc{Certified}
        \bigl(i,\{z_j,\ell_j,U_j,\Xi_j\}_{j=1}^K,
        \Call{Acct}{t}\bigr)\)
    \EndIf

    \For{each family \(i\)}
        \State
        \(\mathcal S_i\gets
        \{I\in\mathcal C_i(h):
        U_{\rm cell}(I)\ge\underline M_i^g\}\)
        \State
        \(\mathcal C_i(h+1)\gets
        \Call{FreshChildren}{\mathcal S_i}\)
    \EndFor
    \State \(h\gets h+1\)
\EndWhile

\Statex
\Statex \textbf{Subroutine: resolve one dyadic level}

\Procedure{ResolveLevel}{\(h\)}
    \For{each family \(i\)}
        \State
        \(a_{i,h}\gets
        \min\{1,\mathcal L_i^g\rho_h\}\)
        \State mark every \(I\in\mathcal C_i(h)\) unresolved
    \EndFor
    \State \(r\gets0\)

    \While{some active cell is unresolved}
        \State \(n_r\gets2^{r+1}\)

        \For{each unresolved pair \((i,I)\) in fixed round-robin order}
            \While{\(m_I<n_r\)}
                \If{\(t=B\)}
                    \State \Return \(\textsc{Fail}\)
                \EndIf
                \State deploy \((i,c_I)\) as a designated source
                \State tag the source as certification-owned
                \State \(m_I\gets m_I+1\), \(t\gets t+1\)
            \EndWhile
        \EndFor

        \State freeze all active sets
        \For{\(u=1,\ldots,w\)}
            \If{\(t=B\)}
                \State \Return \(\textsc{Fail}\)
            \EndIf
            \State deploy the first active center under the fixed order
            \State tag it as a filler source; exclude its feedback
                   from designated estimators
            \State \(t\gets t+1\)
        \EndFor

        \State finalize each new designated source as
        \(B_s^{(w)}\gets\mathbf 1\{Z_s=1,\ D_s\le w\}\)

        \For{each still-unresolved pair \((i,I)\)}
            \State \(n_I\gets m_I\)
            \State recompute \(\widehat g_I\) and \(V_I\)
                   from finalized designated samples at \(c_I\)
            \State
            \(\operatorname{rad}_I\gets
            \VSrad{n_I}{V_I}{\eta_{i,h,I,r}}\)
            \If{\(\operatorname{rad}_I\le a_{i,h}/8\)}
                \State mark \(I\) resolved and freeze its statistics
            \EndIf
        \EndFor
        \State \(r\gets r+1\)
    \EndWhile

    \State \Return \(\textsc{Success}\)
\EndProcedure

\Statex
\Statex \textbf{Subroutine: extract one family certificate}

\Procedure{FamilyCertificate}{\(i,h\)}
    \For{each \(I\in\mathcal C_i(h)\)}
        \State
        \(\mathrm{LCB}(I)\gets
        \max\{0,\widehat g_I-\operatorname{rad}_I\}\)
        \State
        \(U_{\rm cell}(I)\gets
        \min\{1,\widehat g_I+\operatorname{rad}_I+a_{i,h}\}\)
    \EndFor

    \State
    \(\underline M_i^g\gets
    \max_{I\in\mathcal C_i(h)}\mathrm{LCB}(I)\)
    \State
    \(\overline M_i^g\gets
    \max_{I\in\mathcal C_i(h)}U_{\rm cell}(I)\)

    \State choose
    \(I_i^L\in
    \arg\max_{I\in\mathcal C_i(h)}\mathrm{LCB}(I)\)
    using fixed tie-breaking
    \State \(z_i\gets c_{I_i^L}\)
    \State
    \(\xi_i^g\gets
    \min\{1,\max\{0,\overline M_i^g-\underline M_i^g\}\}\)

    \State \Return
    \((z_i,\underline M_i^g,\overline M_i^g,\xi_i^g)\)
\EndProcedure

\Function{Acct}{\(t\)}
    \State \Return
    \((t,\delta_{\rm cert},\mathcal S_{\rm cert})\)
\EndFunction

\end{algorithmic}
\end{algorithm}
